<script>
(function () {
  'use strict';

  var INITIAL_DATA = /*FACTORY_DATA*/null;
  var COLUMNS = ['waiting', 'running', 'decided'];
  var ATTENTION_FLAGS = { 'Antwort': true, 'Frage': true, 'Entscheidung': true };
  var state = {
    data: null,
    tickets: [],
    selectedJobId: null,
    openTicketKey: null
  };

  function asArray(value) {
    return Array.isArray(value) ? value : [];
  }

  function idOf(item, fallback) {
    if (item && item.id !== undefined && item.id !== null && item.id !== '') return String(item.id);
    if (item && item.name !== undefined && item.name !== null && item.name !== '') return String(item.name);
    return fallback || '';
  }

  function titleOf(item, fallback) {
    if (item && item.title) return String(item.title);
    if (item && item.name) return String(item.name);
    if (item && item.id !== undefined && item.id !== null) return String(item.id);
    return fallback || 'Ticket';
  }

  function lastRound(job) {
    var rounds = asArray(job && job.rounds);
    if (rounds.length) return rounds[rounds.length - 1];
    if (job && job.latestRound) return job.latestRound;
    if (job && job.lastRound) return job.lastRound;
    return job || null;
  }

  function decisionOf(round) {
    if (!round) return '';
    var raw = round.decision || round.verdict || round.outcome || round.status || round.state;
    if (raw !== undefined && raw !== null && raw !== '') {
      var value = String(raw).toLowerCase();
      if (value.indexOf('accept') === 0 || value === 'approved' || value === 'ok') return 'accepted';
      if (value.indexOf('reject') === 0 || value === 'denied') return 'rejected';
      return value;
    }
    if (round.accepted === true) return 'accepted';
    if (round.rejected === true) return 'rejected';
    return '';
  }

  function statusOf(job) {
    return String((job && (job.status || job.state || job.phase)) || '').toLowerCase();
  }

  function jobIsPending(job) {
    if (!job) return false;
    if (job.pending === true || job.waiting === true || job.awaiting === true) return true;
    var status = statusOf(job);
    return status === 'pending' || status === 'waiting';
  }

  function jobIsAdvancing(job) {
    if (!job) return false;
    if (job.advancing === true) return true;
    var status = statusOf(job);
    return status === 'advancing' || status === 'running';
  }

  function isSoftwareJob(job) {
    if (!job) return false;
    if (job.software === true) return true;
    var kind = String(job.kind || job.type || job.workpiece || job.category || '').toLowerCase();
    return kind === 'software';
  }

  function isActive(job) {
    if (!job) return false;
    if (job.sourceActive === true || job.active === true) return true;
    var status = statusOf(job);
    return status === 'active' || status === 'running';
  }

  function isQueued(job) {
    if (!job) return false;
    if (job.sourceQueued === true || job.queued === true) return true;
    var status = statusOf(job);
    return status === 'queued' || status === 'waiting';
  }

  function classifyJob(job, index) {
    var id = idOf(job, 'J-' + (index + 1));
    var last = lastRound(job);
    var decision = decisionOf(last);
    var pending = jobIsPending(job);
    var active = isActive(job);
    var queued = isQueued(job);
    var ticket = {
      key: 'job:' + id,
      kind: 'job',
      id: id,
      name: (job && job.name) ? String(job.name) : id,
      title: titleOf(job, id),
      job: job,
      jobId: id,
      flags: [],
      column: '',
      meta: ''
    };

    if (isSoftwareJob(job) && (active || queued)) {
      ticket.column = 'waiting';
      ticket.flags = ['Antwort'];
      ticket.meta = active ? 'Software läuft' : 'Software wartet';
      return ticket;
    }
    if (pending && decision === 'rejected') {
      ticket.column = 'waiting';
      ticket.flags = ['Frage'];
      ticket.meta = 'Runde abgelehnt';
      return ticket;
    }
    if (active) {
      ticket.column = 'running';
      ticket.flags = ['In Arbeit'];
      ticket.meta = 'Läuft';
      return ticket;
    }
    if (pending) {
      ticket.column = 'running';
      ticket.flags = ['Warteschlange'];
      ticket.meta = 'Wartet auf nächste Runde';
      return ticket;
    }
    if (decision === 'accepted' && !jobIsAdvancing(job)) {
      ticket.column = 'decided';
      ticket.flags = ['Angenommen'];
      ticket.meta = 'Angenommen';
      return ticket;
    }
    if (decision === 'rejected') {
      ticket.column = 'decided';
      ticket.flags = ['Abgelehnt'];
      ticket.meta = 'Abgelehnt';
      return ticket;
    }
    return null;
  }

  function buildTickets(data) {
    var tickets = [];
    asArray(data && data.proposals).forEach(function (proposal, index) {
      if (!proposal || proposal.approved) return;
      var id = idOf(proposal, 'P-' + (index + 1));
      tickets.push({
        key: 'proposal:' + id,
        kind: 'proposal',
        id: id,
        name: (proposal.name !== undefined && proposal.name !== null && proposal.name !== '') ? String(proposal.name) : id,
        title: titleOf(proposal, id),
        flags: ['Entscheidung'],
        column: 'waiting',
        jobId: proposal.origin || proposal.jobId || null,
        proposal: proposal,
        meta: proposal.origin ? 'Herkunft: ' + proposal.origin : 'Vorschlag'
      });
    });
    asArray(data && data.jobs).forEach(function (job, index) {
      var ticket = classifyJob(job, index);
      if (ticket) tickets.push(ticket);
    });
    return tickets;
  }

  function waitingTickets() {
    return state.tickets.filter(function (ticket) {
      return ticket.column === 'waiting';
    });
  }

  function boardElement() {
    return document.getElementById('taik-board');
  }

  function mountBoard() {
    var board = boardElement();
    var activity = document.getElementById('taik-activity');
    if (!board || !activity) return;
    if (activity.contains(board)) return;
    activity.appendChild(board);
  }

  function sendHost(name, payload) {
    var handlers = window.webkit && window.webkit.messageHandlers;
    if (handlers && handlers[name] && typeof handlers[name].postMessage === 'function') {
      handlers[name].postMessage(payload);
      return;
    }
    window.dispatchEvent(new CustomEvent('taik:host-message', {
      detail: { name: name, payload: payload }
    }));
  }

  function findHistoryItem(jobId) {
    var board = boardElement();
    var attributes = ['data-job-id', 'data-job', 'data-id'];
    for (var a = 0; a < attributes.length; a += 1) {
      var nodes = document.querySelectorAll('[' + attributes[a] + ']');
      for (var i = 0; i < nodes.length; i += 1) {
        var node = nodes[i];
        if (board && board.contains(node)) continue;
        var value = node.getAttribute(attributes[a]);
        if (value !== null && String(value) === jobId) return node;
      }
    }
    return null;
  }

  function selectJob(jobId) {
    if (jobId === undefined || jobId === null || jobId === '') return;
    var id = String(jobId);
    state.selectedJobId = id;
    if (typeof window.selectJob === 'function') {
      window.selectJob(id);
      return;
    }
    var historyItem = findHistoryItem(id);
    if (historyItem) {
      historyItem.click();
      return;
    }
    window.dispatchEvent(new CustomEvent('taik:select-job', {
      detail: { jobId: id }
    }));
  }

  function makeElement(tag, className, text) {
    var element = document.createElement(tag);
    if (className) element.className = className;
    if (text !== undefined && text !== null) element.textContent = String(text);
    return element;
  }

  function openTicket(ticket) {
    state.openTicketKey = ticket.key;
    if (ticket.jobId) selectJob(ticket.jobId);
    render();
  }

  function renderTicket(ticket) {
    var element = makeElement('article', 'taik-ticket');
    element.setAttribute('data-ticket-key', ticket.key);
    element.setAttribute('data-ticket-kind', ticket.kind);
    element.setAttribute('data-ticket-column', ticket.column);
    element.setAttribute('tabindex', '0');
    element.setAttribute('role', 'button');
    if (state.openTicketKey === ticket.key) element.classList.add('is-open');

    var top = makeElement('div', 'taik-ticket-top');
    top.appendChild(makeElement('span', 'taik-ticket-id', '#' + ticket.id));
    element.appendChild(top);

    element.appendChild(makeElement('div', 'taik-ticket-title', ticket.title));

    var flags = makeElement('div', 'taik-ticket-flags');
    asArray(ticket.flags).forEach(function (flag) {
      var flagElement = makeElement('span', 'taik-ticket-flag', flag);
      if (ATTENTION_FLAGS[flag]) flagElement.classList.add('is-attention');
      flags.appendChild(flagElement);
    });
    element.appendChild(flags);

    if (ticket.meta) element.appendChild(makeElement('div', 'taik-ticket-meta', ticket.meta));

    if (ticket.kind === 'proposal') {
      var approve = makeElement('button', 'taik-ticket-action', 'Freigeben');
      approve.type = 'button';
      approve.setAttribute('data-action', 'approve');
      approve.addEventListener('click', function (event) {
        event.stopPropagation();
        sendHost('approveProposal', {
          name: ticket.name || ticket.id
        });
      });
      element.appendChild(approve);
    }

    element.addEventListener('click', function (event) {
      var target = event.target;
      if (target && target.closest && target.closest('.taik-ticket-actions, .taik-ticket-action')) return;
      openTicket(ticket);
    });
    element.addEventListener('keydown', function (event) {
      if (event.key === 'Enter' || event.key === ' ') {
        event.preventDefault();
        openTicket(ticket);
      }
    });

    if (state.openTicketKey === ticket.key) {
      var actions = makeElement('div', 'taik-ticket-actions');

      var comment = document.createElement('textarea');
      comment.className = 'taik-ticket-comment';
      comment.setAttribute('placeholder', 'Kommentar zu diesem Ticket ...');
      comment.addEventListener('click', function (event) {
        event.stopPropagation();
      });
      actions.appendChild(comment);

      var sendComment = makeElement('button', 'taik-ticket-action', 'Kommentar senden');
      sendComment.type = 'button';
      sendComment.setAttribute('data-action', 'comment');
      sendComment.addEventListener('click', function (event) {
        event.stopPropagation();
        var text = comment.value.trim();
        if (!text) return;
        sendHost('commentTicket', {
          name: ticket.name || ticket.id,
          text: text
        });
        comment.value = '';
      });
      actions.appendChild(sendComment);

      element.appendChild(actions);
    }

    return element;
  }

  function renderColumn(columnId) {
    var board = boardElement();
    if (!board) return;
    var list = board.querySelector('[data-ticket-list="' + columnId + '"]');
    if (!list) return;
    list.textContent = '';
    var tickets = state.tickets.filter(function (ticket) {
      return ticket.column === columnId;
    });
    if (!tickets.length) {
      list.appendChild(makeElement('div', 'taik-ticket-empty', 'Keine Tickets'));
      return;
    }
    tickets.forEach(function (ticket) {
      list.appendChild(renderTicket(ticket));
    });
  }

  function renderLead() {
    var waiting = waitingTickets();
    var count = document.getElementById('taik-ticket-count');
    var jump = document.getElementById('taik-ticket-jump');
    if (count) {
      count.textContent = waiting.length === 1
        ? '1 Ticket wartet auf Antwort'
        : waiting.length + ' Tickets warten auf Antwort';
    }
    if (jump) jump.disabled = waiting.length === 0;
  }

  function render() {
    state.tickets = buildTickets(state.data || {});
    var openStillExists = state.tickets.some(function (ticket) {
      return ticket.key === state.openTicketKey;
    });
    if (!openStillExists) state.openTicketKey = null;
    COLUMNS.forEach(renderColumn);
    renderLead();
  }

  function jumpToFirst() {
    var first = waitingTickets()[0];
    if (!first) return;
    state.openTicketKey = first.key;
    if (first.jobId) selectJob(first.jobId);
    render();
    var board = boardElement();
    if (!board) return;
    var nodes = board.querySelectorAll('[data-ticket-key]');
    for (var index = 0; index < nodes.length; index += 1) {
      if (nodes[index].getAttribute('data-ticket-key') === first.key) {
        if (typeof nodes[index].scrollIntoView === 'function') {
          nodes[index].scrollIntoView({ block: 'nearest' });
        }
        break;
      }
    }
  }

  document.addEventListener('click', function (event) {
    var target = event.target;
    if (!target || !target.closest) return;
    var jump = target.closest('#taik-ticket-jump');
    if (!jump || jump.disabled) return;
    jumpToFirst();
  });

  function updateFactory(data) {
    state.data = data || {};
    render();
  }

  window.updateFactory = updateFactory;

  mountBoard();

  if (INITIAL_DATA) {
    updateFactory(INITIAL_DATA);
  } else {
    render();
  }
})();
</script>
