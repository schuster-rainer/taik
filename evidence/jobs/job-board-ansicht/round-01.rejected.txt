<script>
(function () {
  'use strict';

  var INITIAL_DATA = /*FACTORY_DATA*/null;
  var COLUMNS = [
    { id: 'waiting', label: 'Wartet auf Antwort' },
    { id: 'running', label: 'Läuft' },
    { id: 'decided', label: 'Entschieden' }
  ];
  var ATTENTION_FLAGS = { 'Antwort': true, 'Frage': true, 'Entscheidung': true };
  var state = {
    data: null,
    tickets: [],
    selectedJobId: null,
    openTicketKey: null
  };
  var board = document.getElementById('taik-ticket-board');
  if (!board) return;

  function asArray(value) {
    return Array.isArray(value) ? value : [];
  }

  function idOf(item, fallback) {
    if (item && item.id !== undefined && item.id !== null) return String(item.id);
    if (item && item.name !== undefined && item.name !== null) return String(item.name);
    return fallback || '';
  }

  function titleOf(item, fallback) {
    if (item && item.title) return String(item.title);
    if (item && item.name) return String(item.name);
    if (item && item.id !== undefined && item.id !== null) return String(item.id);
    return fallback || 'Ticket';
  }

  function lastRound(job) {
    var rounds = asArray(job && job.rounds);
    if (rounds.length) return rounds[rounds.length - 1];
    if (job && job.latestRound) return job.latestRound;
    if (job && job.lastRound) return job.lastRound;
    return job || null;
  }

  function roundDecision(round) {
    if (!round) return '';
    if (round.decision) return String(round.decision).toLowerCase();
    if (round.status) return String(round.status).toLowerCase();
    if (round.accepted === true) return 'accepted';
    if (round.rejected === true) return 'rejected';
    return '';
  }

  function jobIsPending(job) {
    if (!job) return false;
    return job.pending === true ||
      job.waiting === true ||
      job.status === 'pending' ||
      job.state === 'pending' ||
      job.phase === 'pending';
  }

  function jobIsAdvancing(job) {
    return !!(job && (job.advancing === true || job.status === 'advancing'));
  }

  function isSoftwareJob(job) {
    if (!job) return false;
    return job.kind === 'software' ||
      job.type === 'software' ||
      job.software === true ||
      job.workpiece === 'software' ||
      job.category === 'software';
  }

  function isActive(job) {
    return !!(job && (job.sourceActive === true || job.status === 'active' || job.state === 'active'));
  }

  function isQueued(job) {
    return !!(job && (job.sourceQueued === true || job.status === 'queued' || job.state === 'queued'));
  }

  function classifyJob(job, index) {
    var id = idOf(job, 'J-' + (index + 1));
    var last = lastRound(job);
    var decision = roundDecision(last);
    var pending = jobIsPending(job);
    var active = isActive(job);
    var queued = isQueued(job);
    var base = {
      key: 'job:' + id,
      kind: 'job',
      id: id,
      name: id,
      title: titleOf(job, id),
      job: job,
      jobId: id
    };

    if (isSoftwareJob(job) && (active || queued)) {
      base.column = 'waiting';
      base.flags = ['Antwort'];
      base.meta = active ? 'Software läuft' : 'Software wartet';
      return base;
    }
    if (pending && decision === 'rejected') {
      base.column = 'waiting';
      base.flags = ['Frage'];
      base.meta = 'Runde abgelehnt';
      return base;
    }
    if (active) {
      base.column = 'running';
      base.flags = ['In Arbeit'];
      base.meta = 'Läuft';
      return base;
    }
    if (pending) {
      base.column = 'running';
      base.flags = ['Warteschlange'];
      base.meta = 'Wartet auf nächste Runde';
      return base;
    }
    if (decision === 'accepted' && !jobIsAdvancing(job)) {
      base.column = 'decided';
      base.flags = ['Angenommen'];
      base.meta = 'Angenommen';
      return base;
    }
    if (decision === 'rejected') {
      base.column = 'decided';
      base.flags = ['Abgelehnt'];
      base.meta = 'Abgelehnt';
      return base;
    }
    return null;
  }

  function buildTickets(data) {
    var tickets = [];
    asArray(data && data.proposals).forEach(function (proposal, index) {
      if (!proposal || proposal.approved === true) return;
      var id = idOf(proposal, 'P-' + (index + 1));
      tickets.push({
        key: 'proposal:' + id,
        kind: 'proposal',
        id: id,
        name: proposal.name || id,
        title: titleOf(proposal, id),
        flags: ['Entscheidung'],
        column: 'waiting',
        jobId: proposal.origin || proposal.jobId || null,
        proposal: proposal,
        meta: proposal.origin ? 'Herkunft: ' + proposal.origin : 'Vorschlag'
      });
    });
    asArray(data && data.jobs).forEach(function (job, index) {
      var ticket = classifyJob(job, index);
      if (ticket) tickets.push(ticket);
    });
    return tickets;
  }

  function waitingTickets() {
    return state.tickets.filter(function (ticket) {
      return ticket.column === 'waiting';
    });
  }

  function sendHost(name, payload) {
    var handlers = window.webkit && window.webkit.messageHandlers;
    if (handlers && handlers[name] && typeof handlers[name].postMessage === 'function') {
      handlers[name].postMessage(payload);
      return;
    }
    window.dispatchEvent(new CustomEvent('taik:host-message', {
      detail: { name: name, payload: payload }
    }));
  }

  function selectJob(jobId) {
    if (jobId === undefined || jobId === null || jobId === '') return;
    state.selectedJobId = String(jobId);
    window.dispatchEvent(new CustomEvent('taik:select-job', {
      detail: { jobId: state.selectedJobId }
    }));
    if (typeof window.selectJob === 'function') {
      window.selectJob(state.selectedJobId);
    }
  }

  function makeElement(tag, className, text) {
    var element = document.createElement(tag);
    if (className) element.className = className;
    if (text !== undefined && text !== null) element.textContent = String(text);
    return element;
  }

  function renderTicket(ticket) {
    var article = makeElement('article', 'taik-board-ticket');
    article.setAttribute('data-ticket-key', ticket.key);
    article.setAttribute('tabindex', '0');
    article.setAttribute('role', 'button');
    if (state.openTicketKey === ticket.key) article.classList.add('is-open');

    var top = makeElement('div', 'taik-board-ticket-top');
    top.appendChild(makeElement('span', 'taik-board-ticket-id', '#' + ticket.id));
    article.appendChild(top);

    article.appendChild(makeElement('div', 'taik-board-ticket-title', ticket.title));

    var flags = makeElement('div', 'taik-board-flags');
    asArray(ticket.flags).forEach(function (flag) {
      var flagElement = makeElement('span', 'taik-board-flag', flag);
      if (ATTENTION_FLAGS[flag]) flagElement.classList.add('is-attention');
      flags.appendChild(flagElement);
    });
    article.appendChild(flags);

    if (ticket.meta) article.appendChild(makeElement('div', 'taik-board-meta', ticket.meta));

    function openTicket() {
      state.openTicketKey = ticket.key;
      if (ticket.jobId) selectJob(ticket.jobId);
      render();
    }

    article.addEventListener('click', function (event) {
      if (event.target.closest('.taik-board-ticket-actions')) return;
      openTicket();
    });
    article.addEventListener('keydown', function (event) {
      if (event.key === 'Enter' || event.key === ' ') {
        event.preventDefault();
        openTicket();
      }
    });

    if (state.openTicketKey === ticket.key) {
      var actions = makeElement('div', 'taik-board-ticket-actions');

      if (ticket.kind === 'proposal') {
        var approve = makeElement('button', 'taik-board-action', 'Freigeben');
        approve.type = 'button';
        approve.addEventListener('click', function (event) {
          event.stopPropagation();
          sendHost('approveProposal', {
            name: (ticket.proposal && (ticket.proposal.name || ticket.proposal.id)) || ticket.id
          });
        });
        actions.appendChild(approve);
      }

      var comment = document.createElement('textarea');
      comment.className = 'taik-board-comment';
      comment.placeholder = 'Kommentar zu diesem Ticket ...';
      comment.addEventListener('click', function (event) {
        event.stopPropagation();
      });
      actions.appendChild(comment);

      var sendComment = makeElement('button', 'taik-board-action', 'Kommentar senden');
      sendComment.type = 'button';
      sendComment.addEventListener('click', function (event) {
        event.stopPropagation();
        var text = comment.value.trim();
        if (!text) return;
        sendHost('commentTicket', { name: ticket.name || ticket.id, text: text });
        comment.value = '';
      });
      actions.appendChild(sendComment);

      article.appendChild(actions);
    }

    return article;
  }

  function renderColumn(column) {
    var list = board.querySelector('[data-board-list="' + column.id + '"]');
    if (!list) return;
    list.textContent = '';
    var tickets = state.tickets.filter(function (ticket) {
      return ticket.column === column.id;
    });
    if (!tickets.length) {
      list.appendChild(makeElement('div', 'taik-board-empty', 'Keine Tickets'));
      return;
    }
    tickets.forEach(function (ticket) {
      list.appendChild(renderTicket(ticket));
    });
  }

  function renderLead() {
    var waiting = waitingTickets();
    var count = document.getElementById('taik-board-waiting-count');
    var jump = document.getElementById('taik-board-jump');
    if (count) {
      count.textContent = waiting.length === 1
        ? '1 Ticket wartet auf Antwort'
        : waiting.length + ' Tickets warten auf Antwort';
    }
    if (jump) jump.disabled = waiting.length === 0;
  }

  function render() {
    state.tickets = buildTickets(state.data || {});
    var openStillExists = state.tickets.some(function (ticket) {
      return ticket.key === state.openTicketKey;
    });
    if (!openStillExists) state.openTicketKey = null;
    COLUMNS.forEach(renderColumn);
    renderLead();
  }

  var jumpButton = document.getElementById('taik-board-jump');
  if (jumpButton) {
    jumpButton.addEventListener('click', function () {
      var first = waitingTickets()[0];
      if (!first) return;
      state.openTicketKey = first.key;
      if (first.jobId) selectJob(first.jobId);
      render();
      var nodes = board.querySelectorAll('[data-ticket-key]');
      for (var index = 0; index < nodes.length; index += 1) {
        if (nodes[index].getAttribute('data-ticket-key') === first.key) {
          nodes[index].scrollIntoView({ block: 'nearest' });
          break;
        }
      }
    });
  }

  window.updateFactory = function updateFactory(data) {
    state.data = data || {};
    render();
  };

  if (INITIAL_DATA) {
    window.updateFactory(INITIAL_DATA);
  } else {
    render();
  }
})();
</script>
