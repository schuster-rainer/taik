=== main.rs === use std::collections::HashMap; use std::io::{self, Read, Write}; #[derive(Clone, Debug, PartialEq)] enum Tk { Let, Print, Ident(String), Int(i64), Plus, Minus, Star, Slash, Percent, Eq, Semi, LParen, RParen, Eof, Err(&'static str), } #[derive(Clone, Debug)] struct Token { kind: Tk, line: usize, col: usize, } struct Fail { line: usize, col: usize, code: &'static str, } fn lex(src: &str) -> Vec { let chars: Vec = src.chars().collect(); i += 1; continue; } let (sl, sc) = (line, col); if c.is_ascii_alphabetic() || c == '_' { let mut s = String::new(); while i < chars.len() && (chars[i].is_ascii_alphanumeric() || chars[i] == '_') { s.push(chars[i]); col += 1; } if !ok { toks.push(Token { kind: Tk::Err("LITERAL_RANGE"), line: sl, col: sc }); return toks; } toks.push(Token { kind: Tk::Int(v), line: sl, col: sc }); continue; } let kind = match c { '+' => Tk::Plus, '-' => Tk::Minus, '*' => Tk::Star, '/' => Tk::Slash, '%' => Tk::Percent, '=' => Tk::Eq, ';' => Tk::Semi, '(' => Tk::LParen, ')' => Tk::RParen, _ => { toks.push(Token { kind: Tk::Err("BAD_CHAR"), line: sl, col: sc }); return toks; } }; toks.push(Token { kind, line: sl, col: sc }); i += 1; col += 1; } toks.push(Token { kind: Tk::Eof, line, col }); toks } enum Expr { Int(i64), Var(String, usize, usize), Neg(Box, usize, usize), Bin(char, Box, Box, usize, usize), } enum Stmt { Let(String, usize, usize, Expr), Print(Expr), } struct Parser { toks: Vec, pos: usize, } impl Parser { fn peek(&self) -> &Token { &self.toks[self.pos] } fn err_here(&self) -> Fail { let t = self.peek(); let code = match t.kind { Tk::Err(c) => c, _ => "SYNTAX", }; Fail { line: t.line, col: t.col, code } } fn advance(&mut self) -> Token { let t = self.toks[self.pos].clone(); if self.pos + 1 < self.toks.len() { self.pos += 1; } t } fn expect(&mut self, k: Tk) -> Result { if self.peek().kind == k { Ok(self.advance()) } else { Err(self.err_here()) } } fn program(&mut self) -> Result, Fail> { let mut out = Vec::new(); loop { match self.peek().kind { Tk::Eof => return Ok(out), Tk::Let => { self.advance(); let t = self.peek().clone(); let name = match t.kind { Tk::Ident(ref s) => s.clone(), _ => return Err(self.err_here()), }; self.advance(); self.expect(Tk::Eq)?; let e = self.expr()?; self.expect(Tk::Semi)?; out.push(Stmt::Let(name, t.line, t.col, e)); } Tk::Print => { self.advance(); let e = self.expr()?; self.expect(Tk::Semi)?; out.push(Stmt::Print(e)); } _ => return Err(self.err_here()), } } } fn expr(&mut self) -> Result { let mut l = self.mul()?; loop { let op = match self.peek().kind { Tk::Plus => '+', Tk::Minus => '-', _ => return Ok(l), }; let t = self.advance(); let r = self.mul()?; l = Expr::Bin(op, Box::new(l), Box::new(r), t.line, t.col); } } fn mul(&mut self) -> Result { let mut l = self.unary()?; loop { let op = match self.peek().kind { Tk::Star => '*', Tk::Slash => '/', Tk::Percent => '%', _ => return Ok(l), }; let t = self.advance(); let r = self.unary()?; l = Expr::Bin(op, Box::new(l), Box::new(r), t.line, t.col); } } fn unary(&mut self) -> Result { if self.peek().kind == Tk::Minus { let t = self.advance(); let e = self.unary()?; return Ok(Expr::Neg(Box::new(e), t.line, t.col)); } self.primary() } fn primary(&mut self) -> Result { let t = self.peek().clone(); match t.kind { Tk::Int(v) => { self.advance(); Ok(Expr::Int(v)) } Tk::Ident(s) => { self.advance(); Ok(Expr::Var(s, t.line, t.col)) } Tk::LParen => { self.advance(); let e = self.expr()?; self.expect(Tk::RParen)?; Ok(e) } _ => Err(self.err_here()), } } } fn eval(e: &Expr, env: &HashMap) -> Result { match e { Expr::Int(v) => Ok(*v), Expr::Var(s, l, c) => env.get(s).copied().ok_or(Fail { line: *l, col: *c, code: "UNDEFINED" }), Expr::Neg(x, l, c) => { let v = eval(x, env)?; }), Stmt::Let(name, l, c, e) => eval(e, &env).and_then(|v| { if env.contains_key(name) { Err(Fail { line: *l, col: *c, code: "DUPLICATE" }) } else { env.insert(name.clone(), v); Ok(()) } }), }; if let Err(f) = r { let _ = write!(out, "ERROR {}:{} {}\n", f.line, f.col, f.code); let _ = out.flush(); return 1; } } let _ = out.flush(); 0 } fn main() { let h = std::thread::Builder::new() .stack_size(1 << 30) .spawn(|| { let code = run(); std::process::exit(code); }) .unwrap(); let _ = h.join(); }
